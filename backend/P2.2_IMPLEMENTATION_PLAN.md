### Phase 4: Redis Caching Expansion (Expected: -30ms)

**Duration:** 45 minutes  
**Priority:** 🔴 Critical

#### Changes:

1. **Create Cache Middleware**

   ```javascript
   // middleware/cacheMiddleware.js (NEW FILE)
   const cacheService = require("../services/cacheService");

   const cacheMiddleware = (ttl = 60, keyGenerator = null) => {
     return async (req, res, next) => {
       // Skip cache for non-GET requests
       if (req.method !== "GET") return next();

       const cacheKey = keyGenerator
         ? keyGenerator(req)
         : `cache:${req.originalUrl}`;

       try {
         const cached = await cacheService.get(cacheKey);
         if (cached) {
           res.setHeader("X-Cache", "HIT");
           return res.json(cached);
         }

         // Override res.json to cache response
         const originalJson = res.json.bind(res);
         res.json = (data) => {
           if (res.statusCode === 200) {
             cacheService.set(cacheKey, data, ttl);
           }
           res.setHeader("X-Cache", "MISS");
           return originalJson(data);
         };

         next();
       } catch (error) {
         next(); // Fail gracefully
       }
     };
   };

   module.exports = cacheMiddleware;
   ```

2. **Apply to High-Traffic Endpoints**

   ```javascript
   // routes/categoryRoutes.js
   const cacheMiddleware = require("../middleware/cacheMiddleware");

   router.get("/", cacheMiddleware(60), categoryController.getAllCategories);
   ```

   ```javascript
   // routes/requestRoutes.js
   const cacheMiddleware = require("../middleware/cacheMiddleware");

   // Cache published requests for 30s
   router.get(
     "/published",
     cacheMiddleware(
       30,
       (req) => `requests:published:${req.query.categoryId || "all"}`,
     ),
     requestController.getPublishedRequests,
   );
   ```

3. **Cache Invalidation Strategy**
   ```javascript
   // services/requestService.js
   // After creating/updating request:
   await cacheService.del("requests:published:*");
   ```

### Phase 5: Query Optimization (Expected: -25ms)

**Duration:** 60 minutes  
**Priority:** 🟡 High

#### Changes:

1. **Add Database Indexes**

   ```javascript
   // scripts/addPerformanceIndexes.js (NEW FILE)
   const { sequelize } = require("../sequelize_setup");

   async function addIndexes() {
     try {
       // Purchase Requests indexes
       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_requests_status 
               ON "PurchaseRequests"(status);
           `);

       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_requests_created 
               ON "PurchaseRequests"("createdAt" DESC);
           `);

       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_requests_user 
               ON "PurchaseRequests"("userId");
           `);

       // Price Quotes indexes
       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_quotes_request 
               ON "PriceQuotes"("purchaseRequestId");
           `);

       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_quotes_seller 
               ON "PriceQuotes"("sellerId");
           `);

       // Deals indexes
       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_deals_buyer 
               ON "Deals"("buyerId");
           `);

       await sequelize.query(`
               CREATE INDEX IF NOT EXISTS idx_deals_seller 
               ON "Deals"("sellerId");
           `);

       console.log("✅ All performance indexes created successfully");
     } catch (error) {
       console.error("❌ Error creating indexes:", error);
     }
   }

   addIndexes().then(() => process.exit(0));
   ```
