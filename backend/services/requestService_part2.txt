  static async getPublishedRequests(categoryId = null, filters = {}) {

      order: [['createdAt', 'DESC']],
      limit: filters.limit || 50 // ط­ط¯ ظ…ط¹ظ‚ظˆظ„ ط§ظپطھط±ط§ط¶ظٹ
    });
  }

  static async getRequestDetails(requestId, userId) {
    console.log(`Fetching request details for ID: ${requestId}, User ID: ${userId}`);

    try {
      // ط¥ط¹ط¯ط§ط¯ ط§ظ„ظ€ includes ط§ظ„ط£ط³ط§ط³ظٹط©
      const includes = [
        {
          model: User,
          as: 'user',
          attributes: ['id', 'name', 'subscriptionTier', 'rank', 'businessName']
        },
        {
          model: Category,
          as: 'category',
          attributes: ['id', 'name_ar', 'name_en']
        }
      ];

      // ط¬ظ„ط¨ ط§ظ„ط·ظ„ط¨ ط§ظ„ط£ط³ط§ط³ظٹ ط¨ط¯ظˆظ† quotes ط£ظˆظ„ط§ظ‹
      const request = await PurchaseRequest.findByPk(requestId, {
        include: includes
      });

      if (!request) {
        console.error(`Request not found: ${requestId}`);
        throw new Error('Request not found');
      }

      const plainReq = request.get({ plain: true });
      console.log(`Request found: ${plainReq.id}, Status: ${plainReq.status}`);

      // طھط­ط¯ظٹط¯ طµظ„ط§ط­ظٹط§طھ ط§ظ„ظ…ط³طھط®ط¯ظ…
      const user = userId ? await User.findByPk(userId) : null;
      const isOwner = user && user.id === plainReq.userId;
      const isAdmin = user && (user.role === 'admin' || user.role === 'super_admin');
      const isSeller = user && user.role === 'seller';

      // ط¬ظ„ط¨ ط§ظ„ظ€ quotes ط¨ظ†ط§ط،ظ‹ ط¹ظ„ظ‰ ط§ظ„طµظ„ط§ط­ظٹط§طھ
      let quotes = [];
      const quoteWhere = { purchaseRequestId: requestId };

      if (isOwner || isAdmin) {
        // ط§ظ„ظ…ط§ظ„ظƒ ط£ظˆ ط§ظ„ط£ط¯ظ…ظ†: ظٹط±ظˆظ† ط¬ظ…ظٹط¹ ط§ظ„ظ€ quotes
        quotes = await PriceQuote.findAll({
          where: quoteWhere,
          include: [
            {
              model: User,
              as: 'seller',
              attributes: ['id', 'name', 'businessName', 'rank']
            }
          ],
          order: [['createdAt', 'DESC']]
        });
      } else if (isSeller) {
        // ط§ظ„ط¨ط§ط¦ط¹: ظٹط±ظˆظ† ظپظ‚ط· ط§ظ„ظ€ quotes ط§ظ„ط®ط§طµط© ط¨ظ‡ظ… ظپظٹ ط§ظ„ظ…ط²ط§ط¯ ط§ظ„ط³ط±ظٹ
        if (plainReq.auction_type === 'secret') {
          quoteWhere.sellerId = userId;
        }

        quotes = await PriceQuote.findAll({
          where: quoteWhere,
          include: [
            {
              model: User,
              as: 'seller',
              attributes: ['id', 'name', 'businessName', 'rank']
            }
          ],
          order: [['createdAt', 'DESC']]
        });

        // ظپظٹ ط§ظ„ظ…ط²ط§ط¯ ط§ظ„ط¹ط§ظ…طŒ ط¥ط®ظپط§ط، ظ…ط¹ظ„ظˆظ…ط§طھ ط§ظ„ط¨ط§ط¦ط¹ظٹظ† ط§ظ„ط¢ط®ط±ظٹظ†
        if (plainReq.auction_type === 'public' && quotes.length > 0) {
          quotes = quotes.map(quote => {
            const quoteData = quote.get({ plain: true });
            if (quoteData.sellerId !== userId) {
              // ط¥ط®ظپط§ط، ظ…ط¹ظ„ظˆظ…ط§طھ ط§ظ„ط¨ط§ط¦ط¹ ط§ظ„ط¢ط®ط±
              quoteData.seller = {
                id: null,
                name: 'ط¨ط§ط¦ط¹ ط¢ط®ط±',
                businessName: '---',
                rank: null
              };
              quoteData.amount = null;
              quoteData.notes = 'ط¹ط±ط¶ ظ…ط®ظپظٹ';
            }
            return quoteData;
          });
        }
      }

      // ط¥ط¶ط§ظپط© ط§ظ„ظ€ quotes ظ„ظ„ط·ظ„ط¨
      plainReq.quotes = quotes;
      plainReq.quoteCount = quotes.length;

      // ط¥ط®ظپط§ط، ظ…ط¹ظ„ظˆظ…ط§طھ ط§ظ„ط§طھطµط§ظ„ ط¥ط°ط§ ظ„ظ… ظٹظƒظ† ظ…ط§ظ„ظƒ ط£ظˆ ط£ط¯ظ…ظ†
      if (!isOwner && !isAdmin) {
        if (plainReq.contactNumbers) plainReq.contactNumbers = [];
        if (plainReq.contact_number) plainReq.contact_number = null;
        if (plainReq.user && plainReq.user.email) {
          delete plainReq.user.email;
        }
      }

      console.log(`Returning request with ${plainReq.quotes.length} quotes`);
      return plainReq;

    } catch (error) {
      console.error('Error in getRequestDetails:', error);
      throw error;
    }
  }

  static async transitionRequestStatus(requestId, newStatus, user) {
    const request = await PurchaseRequest.findByPk(requestId, {
      include: [
        { model: User, as: 'user', attributes: ['id', 'name', 'email', 'subscriptionTier'] },
        { model: Category, as: 'category', attributes: ['id', 'name_ar', 'name_en'] }
      ]
    });

    if (!request) throw new Error('Request not found');

    const currentStatus = request.status;

    // ط¥ط°ط§ ظƒط§ظ† Admin ط£ظˆ Super Admin
    if (user.role === 'admin' || user.role === 'super_admin') {
      await request.update({ status: newStatus });
      return request;
    }

    // طھط­ظ‚ظ‚ ظ…ظ† ط§ظ„ط§ظ†طھظ‚ط§ظ„ط§طھ ط§ظ„ظ…ط³ظ…ظˆط­ط©
    const allowedTransitions = STATUS_TRANSITIONS[currentStatus];
    if (!allowedTransitions || !allowedTransitions.includes(newStatus)) {
      throw new Error(
        `â‌Œ FORBIDDEN STATUS TRANSITION: Cannot change request status from "${currentStatus}" to "${newStatus}".`
      );
    }

    // طھط­ظ‚ظ‚ ظ…ظ† ط§ظ„ظ…ظ„ظƒظٹط© 
    if (request.userId !== user.id) {
      throw new Error('UNAUTHORIZED: Only the request owner can change its status');
    }

    // ط­ط§ظ„ط© ظ‚ط¨ظˆظ„ ط§ظ„ط·ظ„ط¨
    if (newStatus === 'accepted') {
      const acceptedQuote = await PriceQuote.findOne({
        where: { purchaseRequestId: requestId, status: 'accepted' },
        include: [{ model: User, as: 'seller' }]
      });

      if (!acceptedQuote) {
        throw new Error('Cannot accept request: No accepted quote found. You must accept a quote first.');
      }

      // ط¥ظ†ط´ط§ط، ظپط§طھظˆط±ط© ط¥ظ„ظƒطھط±ظˆظ†ظٹط©
      const invoiceData = {
        invoiceNumber: `INV-${Date.now()}-${Math.floor(Math.random() * 1000)}`,
        date: new Date(),
        buyer: {
          id: request.user.id,
          name: request.user.name,
          email: request.user.email,
          contactNumbers: request.contactNumbers
        },
        seller: {
          id: acceptedQuote.seller.id,
          name: acceptedQuote.seller.name,
          businessName: acceptedQuote.seller.businessName,
          email: acceptedQuote.seller.email
        },
        items: [
          {
            description: request.title,
            quantity: request.quantity,
            unit: request.unit,
            price: acceptedQuote.amount
          }
        ],
        totalAmount: acceptedQuote.amount,
        currency: acceptedQuote.currency,
        terms: acceptedQuote.notes
      };

      await Deal.create({
        purchaseRequestId: requestId,
        priceQuoteId: acceptedQuote.id,
        buyerId: request.userId,
        sellerId: acceptedQuote.sellerId,
        finalAmount: acceptedQuote.amount,
        status: 'processing',
        invoiceData
      });
    }

    await request.update({ status: newStatus });
    return request;
  }

  static validatePrivacySettings(tier, requestData) {
    if (tier === 'free') {

        throw new Error('Hiding personal info requires Plan A or Plan B');
      }
    }
