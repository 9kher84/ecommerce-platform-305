# 2. كود إنشاء JWT

### `backend/controllers/authController.js` (إرسال الـ Token)
```javascript
/**
 * دالة مساعدة لإنشاء وإرسال رمز JWT في الـ Response
 */
const sendTokenResponse = async (user, statusCode, res) => {
  // 1. Create Access Token (Cookie)
  const token = user.getSignedJwtToken();

  // 2. Create Refresh Token (Database & JSON)
  const refreshToken = await user.createRefreshToken();

  // خيارات الكوكي (Cookies) - Security Hardening Day 2
  const options = {
    expires: new Date(Date.now() + 15 * 60 * 1000), // 15 minutes
    httpOnly: true, // Prevent XSS theft
    secure: true, // Send over HTTPS only in production
    sameSite: "None", // Allow Cross-Port on Localhost
  };

  res
    .status(statusCode)
    .cookie("token", token, options)
    .json({
      success: true,
      token, // 🔥 Access Token
      refreshToken, // 🔥 New: Refresh Token
      user: {
        id: user.id,
        name: user.name,
        email: user.email,
        role: user.role,
        subscriptionTier: user.subscriptionTier,
        isActive: user.isActive,
      },
    });
};
```

### `backend/models/User.js` (توليد الـ JWT و `expiresIn`)
```javascript
  User.prototype.getSignedJwtToken = function () {
    const secret = process.env.JWT_SECRET || "supersecret";
    console.log("DEBUG: Signing with secret:", secret);
    const token = jwt.sign(
      {
        id: this.id,
        role: this.role,
        jti: uuidv4(),
      },
      secret,
      { expiresIn: "15m" }, // ⬅️ قيمة expiresIn المحددة هنا
    );
    console.log("DEBUG: Generated Token:", token);
    return token;
  };
```
